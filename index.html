            updateThemeIcons(isDark);
        }

        themeToggleBtn.addEventListener('click', toggleTheme);
        mobileThemeToggleBtn.addEventListener('click', toggleTheme);

        // --- 4. Google 試算表表單提交邏輯 (避開 CORS) ---
        // 已更新為您提供的 Google Apps Script 網頁應用程式 URL
        const GAS_WEB_APP_URL = "https://script.google.com/macros/s/AKfycbxntx56UKWf81l1QZ3rD-xRI_E85RySLrYmYlv99nuGO9wO8o9le5FrwJVQkEzPjyqk/exec";

        const contactForm = document.getElementById('contact-form');
        const submitBtn = document.getElementById('submit-btn');
        const formStatus = document.getElementById('form-status');

        contactForm.addEventListener('submit', function(e) {
            e.preventDefault();

            // 檢查 URL 是否替換
            if (GAS_WEB_APP_URL === "YOUR_GOOGLE_APPS_SCRIPT_WEB_APP_URL") {
                showStatus("請先設置 Apps Script 部署 URL 網址。", "error");
                return;
            }

            // 按鈕顯示載入狀態
            submitBtn.disabled = true;
            submitBtn.innerHTML = '<i class="fa-solid fa-spinner animate-spin"></i><span>傳送中...</span>';
            hideStatus();

            const payload = {
                name: document.getElementById('form-name').value.trim(),
                email: document.getElementById('form-email').value.trim(),
                message: document.getElementById('form-message').value.trim()
            };

            // 使用 fetch 發送 POST，以 no-cors 模式避免瀏覽器跨網域攔截
            fetch(GAS_WEB_APP_URL, {
                method: 'POST',
                mode: 'no-cors',
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify(payload)
            })
            .then(() => {
                showStatus("訊息已成功傳送！我們會盡快與您聯繫。", "success");
                contactForm.reset();
            })
            .catch(error => {
                console.error("Form Submit Error:", error);
                showStatus("傳送失敗，請稍後再試或直接 Email 聯繫。", "error");
            })
            .finally(() => {
                submitBtn.disabled = false;
                submitBtn.innerHTML = '<i class="fa-solid fa-paper-plane"></i><span>送出訊息</span>';
            });
        });

        function showStatus(msg, type) {
            formStatus.textContent = msg;
            formStatus.classList.remove('hidden', 'bg-emerald-100', 'text-emerald-800', 'dark:bg-emerald-950/80', 'dark:text-emerald-300', 'bg-red-100', 'text-red-800', 'dark:bg-red-950/80', 'dark:text-red-300');
            
            if (type === 'success') {
                formStatus.classList.add('bg-emerald-100', 'text-emerald-800', 'dark:bg-emerald-950/80', 'dark:text-emerald-300');
            } else {
                formStatus.classList.add('bg-red-100', 'text-red-800', 'dark:bg-red-950/80', 'dark:text-red-300');
            }
        }

        function hideStatus() {
            formStatus.classList.add('hidden');
        }
    </script>
